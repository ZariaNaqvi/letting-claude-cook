The script reads only the first table of the file and stops at the first non-monthly line, so the later blocks are never touched. It sets any -99.99 or -999 value to missing, and a month with a missing component gets no output row.

```python
import csv
import re
import numpy as np
import pandas as pd

rows = []
header = None
in_block = False
with open("6_Portfolios_ME_Prior_12_2.csv", newline="") as f:
    for line in f:
        s = line.strip()
        if not in_block:
            if s.startswith(",") and "SMALL" in s and "LoPRIOR" in s:
                header = [h.strip() for h in s.split(",")]
                in_block = True
            continue
        parts = [p.strip() for p in s.split(",")]
        if parts and re.fullmatch(r"\d{6}", parts[0]):
            rows.append(parts)
        else:
            if rows:
                break
            if s == "":
                continue
            break

df = pd.DataFrame(rows, columns=header[:len(rows[0])])
df = df.rename(columns={df.columns[0]: "date"})
df["date"] = df["date"].astype(int)
for c in df.columns[1:]:
    df[c] = pd.to_numeric(df[c], errors="coerce")
    df.loc[df[c] <= -99.0, c] = np.nan

val = 0.5 * (df["SMALL HiPRIOR"] + df["BIG HiPRIOR"]) - 0.5 * (df["SMALL LoPRIOR"] + df["BIG LoPRIOR"])
out = pd.DataFrame({"date": df["date"], "value": val}).dropna()
out.to_csv("output.csv", index=False)
```
