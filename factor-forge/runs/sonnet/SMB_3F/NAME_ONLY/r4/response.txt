This script reads only the first section of the file, "Average Value Weighted Returns -- Monthly", and stops at the first non-data line after it. For each month it computes SMB as the average of the three small portfolios minus the average of the three big portfolios, the same formula Ken French uses for the 3-factor SMB. It skips any month with a missing value (-99.99 or -999). I have not run it against the full file.

```python
import re
import pandas as pd

rows = []
in_section = False
started = False
with open("6_Portfolios_2x3.csv", "r", errors="replace") as f:
    for line in f:
        s = line.strip()
        if not in_section:
            if "Average Value Weighted Returns" in s and "Monthly" in s:
                in_section = True
            continue
        if not s:
            if started:
                break
            continue
        parts = [p.strip() for p in s.split(",")]
        if re.fullmatch(r"\d{6}", parts[0]) and len(parts) >= 7:
            started = True
            try:
                vals = [float(x) for x in parts[1:7]]
            except ValueError:
                continue
            rows.append((int(parts[0]), vals))
        else:
            if started:
                break

out = []
for d, v in rows:
    if any(x <= -99.99 or x == -999 for x in v):
        continue
    small = (v[0] + v[1] + v[2]) / 3.0
    big = (v[3] + v[4] + v[5]) / 3.0
    out.append((d, round(small - big, 6)))

pd.DataFrame(out, columns=["date", "value"]).to_csv("output.csv", index=False)
```
