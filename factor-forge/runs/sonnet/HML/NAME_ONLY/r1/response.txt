HML is computed as the average of the two high-BM portfolios minus the average of the two low-BM portfolios, using the value-weighted monthly block of the file: ½(Small HiBM + Big HiBM) − ½(Small LoBM + Big LoBM). I have not run this against the full file.

```python
import re
import numpy as np
import pandas as pd

rows = []
in_block = False
with open("6_Portfolios_2x3.csv", "r") as f:
    for line in f:
        s = line.strip()
        if not in_block:
            if "Average Value Weighted Returns" in s and "Monthly" in s:
                in_block = True
            continue
        if not s:
            if rows:
                break
            continue
        parts = [p.strip() for p in s.split(",")]
        if re.fullmatch(r"\d{6}", parts[0]):
            try:
                vals = [float(x) for x in parts[1:7]]
            except ValueError:
                break
            rows.append([int(parts[0])] + vals)
        else:
            if rows:
                break

df = pd.DataFrame(rows, columns=["date", "sl", "sm", "sh", "bl", "bm", "bh"])
for c in ["sl", "sm", "sh", "bl", "bm", "bh"]:
    df.loc[(df[c] <= -99.0), c] = np.nan
df["value"] = (df["sh"] + df["bh"]) / 2 - (df["sl"] + df["bl"]) / 2
out = df[["date", "value"]].dropna()
out["date"] = out["date"].astype(int)
out.to_csv("output.csv", index=False)
```
