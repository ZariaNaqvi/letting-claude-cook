This script parses the first "Average Value Weighted Returns -- Monthly" block from each file, computes each 2x3 SMB as the small-portfolio mean minus the big-portfolio mean, and averages the three. It keeps only months present in all three files and writes `output.csv`.

```python
import re
import numpy as np
import pandas as pd

FILES = [
    "6_Portfolios_2x3.csv",
    "6_Portfolios_ME_OP_2x3.csv",
    "6_Portfolios_ME_INV_2x3.csv",
]
MARKER = "Average Value Weighted Returns -- Monthly"


def read_vw_monthly(path):
    with open(path, encoding="latin-1") as f:
        lines = f.read().splitlines()

    start = None
    for i, line in enumerate(lines):
        if MARKER.lower() in line.lower():
            start = i
            break
    if start is None:
        raise ValueError(f"Marker not found in {path}")

    j = start + 1
    while j < len(lines) and lines[j].strip() == "":
        j += 1
    header = [h.strip() for h in lines[j].split(",")]
    cols = header[1:]
    j += 1

    rows = []
    while j < len(lines):
        parts = [p.strip() for p in lines[j].split(",")]
        if parts and re.fullmatch(r"\d{6}", parts[0]):
            vals = []
            for p in parts[1:1 + len(cols)]:
                try:
                    v = float(p)
                except ValueError:
                    v = np.nan
                if v <= -99:  # -99.99 / -999 missing codes
                    v = np.nan
                vals.append(v)
            while len(vals) < len(cols):
                vals.append(np.nan)
            rows.append([int(parts[0])] + vals)
        else:
            if rows:
                break  # end of the first block
        j += 1

    df = pd.DataFrame(rows, columns=["date"] + cols)
    df["date"] = df["date"].astype(int)
    df = df.set_index("date")

    # Identify small / big portfolios by name; fall back to position
    small = [c for c in cols if c.upper().startswith("SMALL") or c.upper().startswith("ME1")]
    big = [c for c in cols if c.upper().startswith("BIG") or c.upper().startswith("ME2")]
    if len(small) != 3 or len(big) != 3:
        small = cols[:3]
        big = cols[3:6]

    smb = df[small].mean(axis=1) - df[big].mean(axis=1)
    return smb.rename("smb")


series = [read_vw_monthly(f) for f in FILES]

combined = pd.concat(series, axis=1, join="inner").dropna()
combined = combined.sort_index()

smb = combined.mean(axis=1)

out = pd.DataFrame({
    "date": combined.index.astype(int),
    "value": smb.values,
})
out.to_csv("output.csv", index=False)
```
