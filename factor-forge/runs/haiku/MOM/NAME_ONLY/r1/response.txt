The factor is the average of the two High-prior-return portfolios minus the average of the two Low-prior-return portfolios, using the value-weighted returns from the first (Average Value Weighted Returns, Monthly) table. The script reads only that block, treats -99.99 and -999 as missing, and skips any month where a needed portfolio is missing.

```python
import re
import math
import csv

INPUT = "6_Portfolios_ME_Prior_12_2.csv"
OUTPUT = "output.csv"


def parse_monthly_value_weighted(path):
    """Parse the first monthly block (Average Value Weighted Returns -- Monthly).

    Column order in the file:
      0: SMALL LoPRIOR, 1: ME1 PRIOR2, 2: SMALL HiPRIOR,
      3: BIG LoPRIOR,   4: ME2 PRIOR2, 5: BIG HiPRIOR
    """
    rows = []
    started = False
    date_re = re.compile(r"\d{6}")
    with open(path, "r", encoding="latin-1", errors="ignore") as f:
        for line in f:
            parts = [p.strip() for p in line.strip().split(",")]
            if len(parts) == 7 and date_re.fullmatch(parts[0]):
                started = True
                vals = []
                for p in parts[1:]:
                    try:
                        v = float(p)
                    except ValueError:
                        v = math.nan
                    if v == -99.99 or v == -999.0:
                        v = math.nan
                    vals.append(v)
                rows.append((int(parts[0]), vals))
            elif started:
                # first monthly block ended (blank line / next section / footer)
                break
    return rows


def main():
    rows = parse_monthly_value_weighted(INPUT)

    # Dedupe by date (keep first occurrence) and sort
    seen = {}
    for d, vals in rows:
        if d not in seen:
            seen[d] = vals
    dates = sorted(seen)

    out = []
    for d in dates:
        v = seen[d]
        small_lo, small_hi = v[0], v[2]
        big_lo, big_hi = v[3], v[5]
        needed = [small_lo, small_hi, big_lo, big_hi]
        if any(math.isnan(x) for x in needed):
            continue
        mom = 0.5 * (small_hi + big_hi) - 0.5 * (small_lo + big_lo)
        out.append((d, mom))

    with open(OUTPUT, "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["date", "value"])
        for d, m in out:
            w.writerow([d, repr(float(m))])


if __name__ == "__main__":
    main()
```
