The script reads the first table ('Average Value Weighted Returns -- Monthly') from each of the three files. It does the following:

- **Picking portfolios:** small and big portfolios are chosen by header name. Small is `SMALL*` or `ME1*` and big is `BIG*` or `ME2*`. If the names don't match, it falls back to column position: the first three columns are small and the last three are big.
- **Missing values:** `-99.99` and `-999` are treated as missing.
- **Combining files:** only months present in all three files are kept. Any month where SMB can't be computed is dropped.

```python
import re
import numpy as np
import pandas as pd

FILES = ["6_Portfolios_2x3.csv", "6_Portfolios_ME_OP_2x3.csv", "6_Portfolios_ME_INV_2x3.csv"]


def read_first_vw_monthly(path):
    with open(path, "r", encoding="latin-1") as f:
        lines = f.read().splitlines()
    start = None
    for i, ln in enumerate(lines):
        if "Average Value Weighted Returns" in ln and "Monthly" in ln:
            start = i
            break
    if start is None:
        raise ValueError(f"VW monthly table not found in {path}")
    # header line: next non-empty line
    j = start + 1
    while j < len(lines) and lines[j].strip() == "":
        j += 1
    header = [h.strip() for h in lines[j].split(",")][1:]
    rows = []
    k = j + 1
    while k < len(lines):
        ln = lines[k].strip()
        if ln == "":
            if rows:
                break
            k += 1
            continue
        parts = [p.strip() for p in ln.split(",")]
        if not re.fullmatch(r"\d{6}", parts[0]):
            break
        vals = []
        for p in parts[1:1 + len(header)]:
            try:
                v = float(p)
            except ValueError:
                v = np.nan
            if v <= -99.99 + 1e-9 and (abs(v + 99.99) < 1e-6 or abs(v + 999) < 1e-6):
                v = np.nan
            vals.append(v)
        while len(vals) < len(header):
            vals.append(np.nan)
        rows.append([int(parts[0])] + vals)
        k += 1
    df = pd.DataFrame(rows, columns=["date"] + header).set_index("date")
    return df


def smb_component(df):
    cols = list(df.columns)
    small = [c for c in cols if c.upper().startswith("SMALL") or c.upper().startswith("ME1")]
    big = [c for c in cols if c.upper().startswith("BIG") or c.upper().startswith("ME2")]
    if len(small) != 3 or len(big) != 3:
        small, big = cols[:3], cols[3:6]
    return df[small].mean(axis=1, skipna=False) - df[big].mean(axis=1, skipna=False)


comps = []
for fn in FILES:
    d = read_first_vw_monthly(fn)
    comps.append(smb_component(d).rename(fn))

all_c = pd.concat(comps, axis=1, join="inner")
smb = all_c.mean(axis=1, skipna=False).dropna().sort_index()

out = pd.DataFrame({"date": smb.index.astype(int), "value": smb.values})
out.to_csv("output.csv", index=False)
```
