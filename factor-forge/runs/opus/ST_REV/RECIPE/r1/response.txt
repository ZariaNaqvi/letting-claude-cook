The script reads only the first table, 'Average Value Weighted Returns -- Monthly', and computes 0.5 × (SMALL LoPRIOR + BIG LoPRIOR) − 0.5 × (SMALL HiPRIOR + BIG HiPRIOR) for each month.

- **Locating the table:** columns are matched by header name, not by position. Parsing stops at the first line that is not a `YYYYMM` row, so later tables are never read.
- **Missing data:** values of -99.99 or -999 are treated as missing. Any month with a missing input is dropped from the output.

```python
import pandas as pd
import numpy as np

FNAME = "6_Portfolios_ME_Prior_1_0.csv"

with open(FNAME, "r", encoding="latin-1") as f:
    lines = [l.rstrip("\r\n") for l in f]

# locate first block
start = None
for i, l in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in l:
        start = i
        break
if start is None:
    raise RuntimeError("Monthly VW block not found")

# header line: next non-empty line
j = start + 1
while lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]

rows = []
k = j + 1
while k < len(lines):
    l = lines[k].strip()
    if l == "":
        break
    parts = [p.strip() for p in l.split(",")]
    d = parts[0]
    if not (len(d) == 6 and d.isdigit()):
        break
    vals = []
    for p in parts[1:]:
        try:
            v = float(p)
        except ValueError:
            v = np.nan
        if v <= -99.99 or v == -999:
            v = np.nan
        vals.append(v)
    rows.append([int(d)] + vals)
    k += 1

df = pd.DataFrame(rows, columns=["date"] + cols)

def find(name):
    for c in cols:
        if c.replace(" ", "").upper() == name.replace(" ", "").upper():
            return c
    raise KeyError(name)

sl = find("SMALL LoPRIOR")
sh = find("SMALL HiPRIOR")
bl = find("BIG LoPRIOR")
bh = find("BIG HiPRIOR")

df["value"] = 0.5 * (df[sl] + df[bl]) - 0.5 * (df[sh] + df[bh])
out = df[["date", "value"]].dropna().copy()
out["date"] = out["date"].astype(int)
out = out.sort_values("date").drop_duplicates("date")
out["value"] = out["value"].round(6)
out.to_csv("output.csv", index=False)
```
