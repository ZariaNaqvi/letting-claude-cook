ST_Rev is built from the 2×3 size / prior-month-return (month −1) sorts:

ST_Rev = ½ (Small LoPRIOR + Big LoPRIOR) − ½ (Small HiPRIOR + Big HiPRIOR)

The script:
- Reads only the first block, "Average Value Weighted Returns -- Monthly".
- Treats -99.99 and -999 as missing and drops any month where a needed portfolio is missing.
- Leaves the values unrounded, apart from trimming float noise.

```python
import re
import numpy as np
import pandas as pd

FN = "6_Portfolios_ME_Prior_1_0.csv"

with open(FN, "r", encoding="latin-1") as f:
    lines = [l.rstrip("\r\n") for l in f]

# locate the monthly value-weighted block
start = None
for i, l in enumerate(lines):
    if "Average Value Weighted Returns" in l and "Monthly" in l:
        start = i
        break
if start is None:
    raise RuntimeError("Monthly VW block not found")

# header line: first non-empty line after the title
j = start + 1
while lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]

rows = []
k = j + 1
while k < len(lines):
    l = lines[k].strip()
    if l == "":
        if rows:
            break
        k += 1
        continue
    parts = [p.strip() for p in l.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    vals = []
    for p in parts[1:1 + len(cols)]:
        try:
            v = float(p)
        except ValueError:
            v = np.nan
        if v <= -99.99 + 1e-9 or v == -999:
            v = np.nan
        vals.append(v)
    rows.append([int(parts[0])] + vals)
    k += 1

df = pd.DataFrame(rows, columns=["date"] + cols)

def find(pred):
    for c in cols:
        u = c.upper().replace(" ", "")
        if pred(u):
            return c
    raise RuntimeError("column not found")

small_lo = find(lambda u: ("SMALL" in u or "ME1" in u) and "LO" in u)
small_hi = find(lambda u: ("SMALL" in u or "ME1" in u) and "HI" in u)
big_lo = find(lambda u: ("BIG" in u or "ME2" in u) and "LO" in u)
big_hi = find(lambda u: ("BIG" in u or "ME2" in u) and "HI" in u)

val = 0.5 * (df[small_lo] + df[big_lo]) - 0.5 * (df[small_hi] + df[big_hi])
out = pd.DataFrame({"date": df["date"].astype(int), "value": val})
out = out.dropna().sort_values("date").drop_duplicates("date")
out["value"] = out["value"].round(6)
out.to_csv("output.csv", index=False)
```
