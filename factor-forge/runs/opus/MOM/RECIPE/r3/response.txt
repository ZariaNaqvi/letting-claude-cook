The script reads only the first block, "Average Value Weighted Returns -- Monthly". It finds that block's title line and uses the header row right after it. It keeps reading rows that start with a 6-digit date and stops at the first line that doesn't. Columns are looked up by name, not by position.

Momentum is calculated as `0.5*(SMALL HiPRIOR + BIG HiPRIOR) - 0.5*(SMALL LoPRIOR + BIG LoPRIOR)`. The missing-data codes -99.99 and -999 are treated as missing. Any month where one of the four inputs is missing is left out of the output, so not every month in the file is guaranteed a row.

```python
import re
import numpy as np
import pandas as pd

FNAME = "6_Portfolios_ME_Prior_12_2.csv"

with open(FNAME, "r", encoding="latin-1") as f:
    lines = f.read().splitlines()

# locate the first value-weighted monthly block
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in ln:
        start = i
        break
if start is None:
    raise RuntimeError("Value-weighted monthly block not found")

# header line: next non-empty line after the title
j = start + 1
while j < len(lines) and lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]

rows = []
k = j + 1
date_re = re.compile(r"^\s*(\d{6})\s*,")
while k < len(lines):
    ln = lines[k]
    m = date_re.match(ln)
    if not m:
        break
    parts = [p.strip() for p in ln.split(",")]
    date = int(parts[0])
    vals = []
    for p in parts[1:1 + len(cols)]:
        try:
            vals.append(float(p))
        except ValueError:
            vals.append(np.nan)
    while len(vals) < len(cols):
        vals.append(np.nan)
    rows.append([date] + vals)
    k += 1

df = pd.DataFrame(rows, columns=["date"] + cols)
for c in cols:
    df[c] = df[c].astype(float)
    df.loc[(df[c] <= -99.99 + 1e-9) | (df[c] == -999), c] = np.nan

def col(name):
    for c in cols:
        if c.upper().replace(" ", "") == name.upper().replace(" ", ""):
            return df[c]
    raise KeyError(name)

sh = col("SMALL HiPRIOR")
bh = col("BIG HiPRIOR")
sl = col("SMALL LoPRIOR")
bl = col("BIG LoPRIOR")

value = 0.5 * (sh + bh) - 0.5 * (sl + bl)

out = pd.DataFrame({"date": df["date"].astype(int), "value": value})
out = out.dropna(subset=["value"])
out = out.drop_duplicates(subset=["date"]).sort_values("date")
out["value"] = out["value"].round(6)
out.to_csv("output.csv", index=False)
```
