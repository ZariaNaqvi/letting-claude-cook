The script reads only the first monthly value-weighted table and computes RMW as 0.5·(SMALL HiOP + BIG HiOP) − 0.5·(SMALL LoOP + BIG LoOP). Months where any of the four inputs is a missing code (-99.99 or -999) are left out, so no value is computed from missing data.

```python
import re
import numpy as np
import pandas as pd

FNAME = "6_Portfolios_ME_OP_2x3.csv"

with open(FNAME, "r", encoding="latin-1") as f:
    lines = [ln.rstrip("\r\n") for ln in f]

# Locate the first value-weighted monthly block
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns" in ln and "Monthly" in ln:
        start = i
        break
if start is None:
    raise RuntimeError("Value weighted monthly block not found")

# Header line is the next non-empty line
j = start + 1
while lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]

rows = []
k = j + 1
while k < len(lines):
    ln = lines[k].strip()
    if ln == "":
        break
    parts = [p.strip() for p in ln.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    vals = []
    for p in parts[1:]:
        try:
            vals.append(float(p))
        except ValueError:
            vals.append(np.nan)
    rows.append([int(parts[0])] + vals)
    k += 1

df = pd.DataFrame(rows, columns=["date"] + cols)
for c in cols:
    df.loc[df[c] <= -99.99, c] = np.nan

def find(name):
    for c in cols:
        if c.replace(" ", "").upper() == name.replace(" ", "").upper():
            return c
    raise KeyError(name)

sh, bh = find("SMALL HiOP"), find("BIG HiOP")
sl, bl = find("SMALL LoOP"), find("BIG LoOP")

df["value"] = 0.5 * (df[sh] + df[bh]) - 0.5 * (df[sl] + df[bl])
out = df[["date", "value"]].dropna().copy()
out["date"] = out["date"].astype(int)
out = out.sort_values("date").drop_duplicates("date")
out["value"] = out["value"].round(6)
out.to_csv("output.csv", index=False)
```
