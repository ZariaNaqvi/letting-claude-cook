HML is built from the value-weighted monthly returns of the six size/book-to-market portfolios:

HML = ½·(Small High + Big High) − ½·(Small Low + Big Low)

The script works as follows:

- **Which block it reads:** it uses only the "Average Value Weighted Returns -- Monthly" block of the file. It stops at the first line that is not a `YYYYMM` row, so the equal-weighted and annual blocks are excluded.
- **Missing data:** the file marks missing values as `-99.99` or `-999`. These are treated as missing, and any month with a missing input is dropped from the output.

```python
import re
import numpy as np
import pandas as pd

FNAME = "6_Portfolios_2x3.csv"

with open(FNAME, "r", encoding="utf-8", errors="replace") as f:
    lines = f.read().splitlines()

# locate the value-weighted monthly section
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns" in ln and "Monthly" in ln:
        start = i
        break
if start is None:
    raise RuntimeError("Value-weighted monthly section not found")

# header line is the next non-empty line
j = start + 1
while j < len(lines) and lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]
j += 1

row_re = re.compile(r"^\s*(\d{6})\s*,")
records = []
while j < len(lines):
    ln = lines[j]
    m = row_re.match(ln)
    if not m:
        break
    parts = [p.strip() for p in ln.split(",")]
    date = int(parts[0])
    vals = []
    for p in parts[1:1 + len(cols)]:
        try:
            v = float(p)
        except ValueError:
            v = np.nan
        if v <= -99.99 + 1e-9 and (abs(v + 99.99) < 1e-6 or abs(v + 999) < 1e-6):
            v = np.nan
        vals.append(v)
    while len(vals) < len(cols):
        vals.append(np.nan)
    records.append([date] + vals)
    j += 1

df = pd.DataFrame(records, columns=["date"] + cols)

def find_col(keys):
    for c in cols:
        cu = c.upper().replace(" ", "")
        if all(k in cu for k in keys):
            return c
    raise RuntimeError(f"Column with {keys} not found in {cols}")

s_lo = find_col(["SMALL", "LOBM"])
s_hi = find_col(["SMALL", "HIBM"])
b_lo = find_col(["BIG", "LOBM"])
b_hi = find_col(["BIG", "HIBM"])

hml = 0.5 * (df[s_hi] + df[b_hi]) - 0.5 * (df[s_lo] + df[b_lo])

out = pd.DataFrame({"date": df["date"].astype(int), "value": hml})
out = out.dropna(subset=["value"])
out = out.drop_duplicates(subset="date").sort_values("date")
out["value"] = out["value"].round(4)
out.to_csv("output.csv", index=False)
```
