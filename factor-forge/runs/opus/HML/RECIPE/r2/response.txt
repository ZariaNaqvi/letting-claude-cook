This script reads only the first table (value-weighted, monthly) and computes HML as 0.5·(Small HiBM + Big HiBM) − 0.5·(Small LoBM + Big LoBM). Months where any of the four needed portfolios is coded as missing (-99.99 or -999) are dropped instead of being used as real returns.

```python
import re
import numpy as np
import pandas as pd

FNAME = "6_Portfolios_2x3.csv"

with open(FNAME, "r", encoding="latin-1") as f:
    lines = [ln.rstrip("\r\n") for ln in f]

# Locate the first block: Average Value Weighted Returns -- Monthly
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns" in ln and "Monthly" in ln:
        start = i
        break
if start is None:
    raise RuntimeError("Value-weighted monthly block not found")

# Header line follows
hdr_idx = start + 1
while lines[hdr_idx].strip() == "":
    hdr_idx += 1
header = [h.strip() for h in lines[hdr_idx].split(",")]
cols = header[1:]

rows = []
for ln in lines[hdr_idx + 1:]:
    s = ln.strip()
    if s == "":
        break
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    vals = []
    for p in parts[1:1 + len(cols)]:
        try:
            vals.append(float(p))
        except ValueError:
            vals.append(np.nan)
    while len(vals) < len(cols):
        vals.append(np.nan)
    rows.append([int(parts[0])] + vals)

df = pd.DataFrame(rows, columns=["date"] + cols)

def find_col(name):
    norm = lambda x: re.sub(r"\s+", " ", x.strip().upper())
    for c in cols:
        if norm(c) == norm(name):
            return c
    raise KeyError(name)

sh = find_col("SMALL HiBM")
bh = find_col("BIG HiBM")
sl = find_col("SMALL LoBM")
bl = find_col("BIG LoBM")

sub = df[["date", sh, bh, sl, bl]].copy()
for c in [sh, bh, sl, bl]:
    sub.loc[(sub[c] <= -99.99) | (sub[c] == -999), c] = np.nan

sub["value"] = 0.5 * (sub[sh] + sub[bh]) - 0.5 * (sub[sl] + sub[bl])
out = sub[["date", "value"]].dropna()
out["date"] = out["date"].astype(int)
out = out.drop_duplicates("date").sort_values("date")
out.to_csv("output.csv", index=False)
```
