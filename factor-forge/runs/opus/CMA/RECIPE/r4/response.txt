The script reads only the first block ('Average Value Weighted Returns -- Monthly') and stops at the first line that isn't a monthly `YYYYMM` row. That end-of-block line is either a blank line or the start of the next block. For each month it computes 0.5·(SMALL LoINV + BIG LoINV) − 0.5·(SMALL HiINV + BIG HiINV).

If any of the four inputs for a month is -99.99 or -999, that month is left out of output.csv rather than written with a made-up value. All months are written if there are no missing values.

```python
import pandas as pd
import numpy as np

FNAME = "6_Portfolios_ME_INV_2x3.csv"

with open(FNAME, "r", encoding="latin-1") as f:
    lines = [ln.rstrip("\r\n") for ln in f]

# locate the first block header
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in ln:
        start = i
        break
if start is None:
    raise RuntimeError("Value-weighted monthly block not found")

# header line: next non-empty line
j = start + 1
while j < len(lines) and lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
cols = header[1:]
j += 1

rows = []
while j < len(lines):
    ln = lines[j].strip()
    if ln == "":
        break
    parts = [p.strip() for p in ln.split(",")]
    d = parts[0]
    if not (d.isdigit() and len(d) == 6):
        break
    vals = []
    for p in parts[1:1 + len(cols)]:
        try:
            vals.append(float(p))
        except ValueError:
            vals.append(np.nan)
    while len(vals) < len(cols):
        vals.append(np.nan)
    rows.append([int(d)] + vals)
    j += 1

df = pd.DataFrame(rows, columns=["date"] + cols)

def find_col(name):
    norm = lambda s: " ".join(s.upper().split())
    for c in cols:
        if norm(c) == norm(name):
            return c
    raise KeyError(name)

sl = find_col("SMALL LoINV")
sh = find_col("SMALL HiINV")
bl = find_col("BIG LoINV")
bh = find_col("BIG HiINV")

use = df[["date", sl, sh, bl, bh]].copy()
for c in [sl, sh, bl, bh]:
    use.loc[use[c] <= -99.99, c] = np.nan

use["value"] = 0.5 * (use[sl] + use[bl]) - 0.5 * (use[sh] + use[bh])
out = use[["date", "value"]].dropna()
out = out.drop_duplicates("date").sort_values("date")
out["date"] = out["date"].astype(int)
out.to_csv("output.csv", index=False)
```
