```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_ME_INV_2x3.csv"

with open(fname, "r", encoding="utf-8", errors="replace") as f:
    lines = f.read().splitlines()

# Find the first monthly value-weighted block
start = None
for i, line in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in line:
        start = i
        break
if start is None:
    raise RuntimeError("Could not find monthly VW block")

# Header line is the next line starting with a comma
hdr_idx = None
for j in range(start + 1, min(start + 10, len(lines))):
    if lines[j].strip().startswith(","):
        hdr_idx = j
        break
if hdr_idx is None:
    raise RuntimeError("Header not found")

header = [h.strip() for h in lines[hdr_idx].split(",")]
header[0] = "date"

rows = []
for line in lines[hdr_idx + 1:]:
    s = line.strip()
    if not s:
        break
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    if len(parts) != len(header):
        break
    try:
        vals = [float(p) for p in parts[1:]]
    except ValueError:
        break
    rows.append([int(parts[0])] + vals)

df = pd.DataFrame(rows, columns=header)
df = df.replace([-99.99, -999], np.nan)

def col(name):
    for c in df.columns:
        if c.lower().replace(" ", "") == name.lower().replace(" ", ""):
            return df[c]
    raise KeyError(name)

small_lo = col("SMALL LoINV")
big_lo = col("BIG LoINV")
small_hi = col("SMALL HiINV")
big_hi = col("BIG HiINV")

cma = 0.5 * (small_lo + big_lo) - 0.5 * (small_hi + big_hi)

out = pd.DataFrame({"date": df["date"].astype(int), "value": cma})
out = out.dropna(subset=["value"])
out.to_csv("output.csv", index=False)
```

The script locates the first "Average Value Weighted Returns -- Monthly" block, parses only the YYYYMM-dated rows until the block ends, and computes CMA = ½(SMALL LoINV + BIG LoINV) − ½(SMALL HiINV + BIG HiINV) in percent.
