```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_ME_INV_2x3.csv"

with open(fname, "r") as f:
    lines = f.read().splitlines()

# Locate the monthly value-weighted section
start = None
for i, line in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in line:
        start = i
        break
if start is None:
    # fallback: first header line beginning with a comma
    for i, line in enumerate(lines):
        if line.startswith(","):
            start = i - 1
            break

# Header line is the next line starting with a comma
hdr_idx = start + 1
while hdr_idx < len(lines) and not lines[hdr_idx].strip().startswith(","):
    hdr_idx += 1
header = [h.strip() for h in lines[hdr_idx].split(",")]

rows = []
for line in lines[hdr_idx + 1:]:
    s = line.strip()
    if s == "":
        break
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    try:
        vals = [float(p) for p in parts[1:]]
    except ValueError:
        break
    rows.append([int(parts[0])] + vals)

df = pd.DataFrame(rows, columns=["date"] + header[1:])
df = df.set_index("date")
df = df.replace([-99.99, -999.0], np.nan)

cols = list(df.columns)
# Identify small/big and Lo/Hi INV columns
def find_col(size_kw, inv_kw):
    for c in cols:
        cl = c.lower()
        if size_kw in cl and inv_kw in cl:
            return c
    return None

small_lo = find_col("small", "loinv") or find_col("me1", "inv1") or cols[0]
small_hi = find_col("small", "hiinv") or find_col("me1", "inv3") or cols[2]
big_lo = find_col("big", "loinv") or find_col("me2", "inv1") or cols[3]
big_hi = find_col("big", "hiinv") or find_col("me2", "inv3") or cols[5]

cma = 0.5 * (df[small_lo] + df[big_lo]) - 0.5 * (df[small_hi] + df[big_hi])

out = pd.DataFrame({"date": df.index.astype(int), "value": cma.values})
out = out.dropna(subset=["value"])
out["value"] = out["value"].round(4)
out.to_csv("output.csv", index=False)
```

CMA is computed as the average of the two conservative (LoINV) portfolios minus the average of the two aggressive (HiINV) portfolios from the monthly value-weighted 2x3 ME×INV portfolios: CMA = ½(Small Lo + Big Lo) − ½(Small Hi + Big Hi).
