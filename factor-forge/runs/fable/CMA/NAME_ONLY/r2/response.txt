```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_ME_INV_2x3.csv"

with open(fname, "r") as f:
    lines = f.read().splitlines()

# Locate the first monthly value-weighted section header line (starts with a comma, lists portfolio names)
start = None
for i, line in enumerate(lines):
    if "Average Value Weighted Returns -- Monthly" in line:
        start = i
        break
if start is None:
    # fallback: first line beginning with a comma
    for i, line in enumerate(lines):
        if line.startswith(","):
            start = i - 1
            break

# Header row is the next line starting with a comma
hdr_idx = start + 1
while hdr_idx < len(lines) and not lines[hdr_idx].startswith(","):
    hdr_idx += 1
header = [h.strip() for h in lines[hdr_idx].split(",")]

rows = []
for line in lines[hdr_idx + 1:]:
    s = line.strip()
    if s == "":
        break
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    rows.append(parts)

df = pd.DataFrame(rows, columns=header)
df.rename(columns={header[0]: "date"}, inplace=True)
df["date"] = df["date"].astype(int)
for c in df.columns[1:]:
    df[c] = pd.to_numeric(df[c], errors="coerce")
df = df.replace([-99.99, -999.0], np.nan)

cols = list(df.columns[1:])
# Columns order: SMALL LoINV, ME1 INV2, SMALL HiINV, BIG LoINV, ME2 INV2, BIG HiINV
small_lo, small_hi = cols[0], cols[2]
big_lo, big_hi = cols[3], cols[5]

cma = 0.5 * (df[small_lo] + df[big_lo]) - 0.5 * (df[small_hi] + df[big_hi])

out = pd.DataFrame({"date": df["date"].astype(int), "value": cma.round(4)})
out = out.dropna(subset=["value"])
out.to_csv("output.csv", index=False)
```

CMA = ½(Small Conservative + Big Conservative) − ½(Small Aggressive + Big Aggressive), using the monthly value-weighted 2x3 ME×INV portfolios (first section of the file only).
