```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_ME_Prior_12_2.csv"

with open(fname, "r", encoding="utf-8", errors="replace") as f:
    lines = f.read().splitlines()

# Locate the first table: "Average Value Weighted Returns -- Monthly"
start = None
for i, line in enumerate(lines):
    if "Average Value Weighted Returns" in line and "Monthly" in line:
        start = i
        break
if start is None:
    # fallback: first header line beginning with a comma
    for i, line in enumerate(lines):
        if line.startswith(","):
            start = i - 1
            break

# Header line is the next line starting with a comma
hdr_idx = start + 1
while hdr_idx < len(lines) and not lines[hdr_idx].strip().startswith(","):
    hdr_idx += 1
header = [h.strip() for h in lines[hdr_idx].split(",")]

rows = []
for line in lines[hdr_idx + 1:]:
    s = line.strip()
    if not s:
        # blank line ends the block (allow until first data seen)
        if rows:
            break
        continue
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        # non-monthly row (e.g., annual YYYY or new block title) -> stop
        if rows:
            break
        continue
    try:
        vals = [float(x) for x in parts[1:]]
    except ValueError:
        break
    rows.append([int(parts[0])] + vals)

df = pd.DataFrame(rows, columns=["date"] + header[1:])

# Treat missing codes as NaN
for c in df.columns[1:]:
    df.loc[df[c] <= -99.0, c] = np.nan

def col(name):
    # robust matching of column names
    for c in df.columns:
        if c.strip().upper() == name.upper():
            return df[c]
    raise KeyError(name)

small_hi = col("SMALL HiPRIOR")
big_hi = col("BIG HiPRIOR")
small_lo = col("SMALL LoPRIOR")
big_lo = col("BIG LoPRIOR")

mom = 0.5 * (small_hi + big_hi) - 0.5 * (small_lo + big_lo)

out = pd.DataFrame({"date": df["date"].astype(int), "value": mom.round(4)})
out = out.dropna(subset=["value"])
out.to_csv("output.csv", index=False)
```

The script parses only the first (value-weighted monthly) block, stops at the first non-YYYYMM row, computes Mom = ½(Small Hi + Big Hi) − ½(Small Lo + Big Lo), and writes `output.csv` with `date,value`.
