```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_ME_Prior_60_13.csv"

with open(fname, "r") as f:
    lines = f.read().splitlines()

# Locate the monthly value-weighted section (first section in the file)
start = None
for i, line in enumerate(lines):
    if "Average Value Weighted Returns" in line and "Monthly" in line:
        start = i
        break
if start is None:
    # fallback: first header line beginning with a comma
    for i, line in enumerate(lines):
        if line.startswith(","):
            start = i - 1
            break

# Find header row (begins with comma) after start
hdr = None
for i in range(start, len(lines)):
    if lines[i].strip().startswith(","):
        hdr = i
        break

cols = [c.strip() for c in lines[hdr].split(",")]
rows = []
for line in lines[hdr + 1:]:
    s = line.strip()
    if not s:
        break
    parts = [p.strip() for p in s.split(",")]
    if not re.fullmatch(r"\d{6}", parts[0]):
        break
    rows.append(parts)

df = pd.DataFrame(rows, columns=cols)
df.rename(columns={cols[0]: "date"}, inplace=True)
df["date"] = df["date"].astype(int)
for c in df.columns[1:]:
    df[c] = pd.to_numeric(df[c], errors="coerce")

# Replace missing codes with NaN
vals = df.columns[1:]
df[vals] = df[vals].replace([-99.99, -999.0], np.nan)

small_lo = df["SMALL LoPRIOR"]
small_hi = df["SMALL HiPRIOR"]
big_lo = df["BIG LoPRIOR"]
big_hi = df["BIG HiPRIOR"]

# LT_Rev = 1/2 (Small Low + Big Low) - 1/2 (Small High + Big High)
ltrev = 0.5 * (small_lo + big_lo) - 0.5 * (small_hi + big_hi)

out = pd.DataFrame({"date": df["date"], "value": ltrev.round(2)})
out = out.dropna(subset=["value"])
out["date"] = out["date"].astype(int)
out.to_csv("output.csv", index=False)
```

The script parses only the first (monthly, value-weighted) section of the file and computes LT_Rev as the average of the two low-prior-return portfolios minus the average of the two high-prior-return portfolios, following Ken French's definition.
