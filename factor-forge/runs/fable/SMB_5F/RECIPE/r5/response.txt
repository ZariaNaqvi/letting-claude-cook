```python
import re
import pandas as pd
import numpy as np

def read_first_monthly_block(fname):
    with open(fname, "r", encoding="utf-8", errors="replace") as f:
        lines = f.read().splitlines()
    # locate the first "Average Value Weighted Returns -- Monthly" block
    start = None
    for i, ln in enumerate(lines):
        if "Average Value Weighted Returns" in ln and "Monthly" in ln:
            start = i
            break
    if start is None:
        raise RuntimeError(f"Block not found in {fname}")
    # header line is the next non-empty line
    j = start + 1
    while j < len(lines) and lines[j].strip() == "":
        j += 1
    header = [h.strip() for h in lines[j].split(",")]
    cols = header[1:]
    rows = []
    k = j + 1
    pat = re.compile(r"^\s*\d{6}\s*,")
    while k < len(lines):
        ln = lines[k]
        if not pat.match(ln):
            break
        parts = [p.strip() for p in ln.split(",")]
        date = int(parts[0])
        vals = []
        for p in parts[1:1 + len(cols)]:
            try:
                v = float(p)
            except ValueError:
                v = np.nan
            if v <= -99.0:
                v = np.nan
            vals.append(v)
        rows.append([date] + vals)
        k += 1
    df = pd.DataFrame(rows, columns=["date"] + cols).set_index("date")
    return df

def smb_component(df):
    cols = list(df.columns)
    small = [c for c in cols if c.upper().startswith("SMALL") or c.upper().startswith("ME1")]
    big = [c for c in cols if c.upper().startswith("BIG") or c.upper().startswith("ME2")]
    if len(small) != 3 or len(big) != 3:
        # fallback: positional (first 3 small, last 3 big)
        small, big = cols[:3], cols[3:6]
    return df[small].mean(axis=1) - df[big].mean(axis=1)

bm = read_first_monthly_block("6_Portfolios_2x3.csv")
op = read_first_monthly_block("6_Portfolios_ME_OP_2x3.csv")
inv = read_first_monthly_block("6_Portfolios_ME_INV_2x3.csv")

s_bm = smb_component(bm)
s_op = smb_component(op)
s_inv = smb_component(inv)

common = s_bm.index.intersection(s_op.index).intersection(s_inv.index)
common = sorted(common)

smb = (s_bm.loc[common] + s_op.loc[common] + s_inv.loc[common]) / 3.0
out = pd.DataFrame({"date": [int(d) for d in common], "value": smb.values})
out = out.dropna(subset=["value"])
out["date"] = out["date"].astype(int)
out = out.sort_values("date").reset_index(drop=True)
out.to_csv("output.csv", index=False, float_format="%.6f")
```

The script parses only the first "Average Value Weighted Returns -- Monthly" block of each file (stopping at the first non-YYYYMM row), computes each file's SMB component as mean(small 3) − mean(big 3), averages the three over the common months (196307 onward), and writes `output.csv` with `date,value` in percent.
