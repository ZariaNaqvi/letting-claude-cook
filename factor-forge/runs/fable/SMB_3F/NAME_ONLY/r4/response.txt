```python
import pandas as pd
import numpy as np
import re

fname = "6_Portfolios_2x3.csv"

with open(fname, "r", encoding="latin-1") as f:
    lines = f.read().splitlines()

# Locate the first monthly value-weighted section
start = None
for i, ln in enumerate(lines):
    if "Average Value Weighted Returns" in ln and "Monthly" in ln:
        start = i
        break
if start is None:
    # fallback: first section is value weighted monthly
    start = 0

# header is next non-empty line after start
j = start + 1
while j < len(lines) and lines[j].strip() == "":
    j += 1
header = [h.strip() for h in lines[j].split(",")]
j += 1

rows = []
pat = re.compile(r"^\s*(\d{6})\s*,")
for k in range(j, len(lines)):
    ln = lines[k]
    if ln.strip() == "":
        break
    m = pat.match(ln)
    if not m:
        break
    parts = [p.strip() for p in ln.split(",")]
    date = int(parts[0])
    vals = []
    for p in parts[1:7]:
        try:
            vals.append(float(p))
        except ValueError:
            vals.append(np.nan)
    rows.append([date] + vals)

cols = ["date", "SL", "SM", "SH", "BL", "BM", "BH"]
df = pd.DataFrame(rows, columns=cols)

# treat missing codes as NaN
for c in cols[1:]:
    df.loc[df[c] <= -99.0, c] = np.nan

small = (df["SL"] + df["SM"] + df["SH"]) / 3.0
big = (df["BL"] + df["BM"] + df["BH"]) / 3.0
df["value"] = small - big

out = df[["date", "value"]].dropna().copy()
out["date"] = out["date"].astype(int)
out["value"] = out["value"].round(4)
out = out.sort_values("date").drop_duplicates("date")
out.to_csv("output.csv", index=False)
```

SMB = 1/3(Small Value + Small Neutral + Small Growth) − 1/3(Big Value + Big Neutral + Big Growth), computed from the monthly value-weighted 2x3 portfolios (first section of the file).
