```python
import re
import pandas as pd
import numpy as np

fname = "6_Portfolios_ME_Prior_1_0.csv"

with open(fname, "r") as f:
    lines = f.readlines()

# Locate the first monthly block (value-weighted) and parse its rows
rows = []
in_block = False
started = False
for line in lines:
    s = line.strip()
    if not in_block:
        if s.startswith("Average Value Weighted Returns -- Monthly"):
            in_block = True
        continue
    # inside block
    if re.match(r"^\d{6},", s):
        parts = [p.strip() for p in s.split(",")]
        if len(parts) >= 7:
            try:
                vals = [float(x) for x in parts[1:7]]
            except ValueError:
                continue
            rows.append([int(parts[0])] + vals)
            started = True
    else:
        if started and s and not s.startswith(","):
            # reached the next section header
            break

cols = ["date", "S_Lo", "S_Med", "S_Hi", "B_Lo", "B_Med", "B_Hi"]
df = pd.DataFrame(rows, columns=cols)

# Treat missing markers as NaN
for c in cols[1:]:
    df.loc[df[c].isin([-99.99, -999.0]), c] = np.nan

# ST_Rev = 1/2 (Small Low + Big Low) - 1/2 (Small High + Big High)
df["value"] = 0.5 * (df["S_Lo"] + df["B_Lo"]) - 0.5 * (df["S_Hi"] + df["B_Hi"])
df = df.dropna(subset=["value"])
df["value"] = df["value"].round(2)

out = df[["date", "value"]].copy()
out["date"] = out["date"].astype(int)
out = out.sort_values("date").drop_duplicates("date")
out.to_csv("output.csv", index=False)
```

ST_Rev is the average of the two low prior-return (1-month) portfolios minus the average of the two high prior-return portfolios, using the value-weighted monthly block of the 2×3 ME × Prior(1-0) file.
